===== JS =====
/* ════════════════ 오프닝 : 봉투 개봉 ════════════════ */
(function openingEnvelope(){
  const intro = $('intro');
  if (!intro) return;

  const opt = CONFIG.intro || {};
  const reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  if (!opt.enabled || reduced) { intro.remove(); return; }

  /* 편지지에 이름과 날짜 */
  $('sealNames').innerHTML =
    `${CONFIG.groom.firstName}<em>&amp;</em>${CONFIG.bride.firstName}`;
  $('sealDate').textContent =
    `${D.year} . ${pad(D.month)} . ${pad(D.day)}  ${DOW_EN[weddingAt.getDay()]}`;

  /* ── 봉랍 : 세 겹(그늘·빛·본체)을 겹쳐 양각으로 ── */
  const motif = opt.seal === 'monogram'
    ? `<text x="50" y="50" text-anchor="middle" dominant-baseline="central"
             font-family="Great Vibes, cursive" font-size="46"
             fill="currentColor" stroke="none">${opt.monogram || 'C&amp;Y'}</text>`
    : `<g fill="currentColor" stroke="currentColor" stroke-linecap="round">
         <path d="M50 74 C50 61 50 47 50 33" stroke-width="2.1" fill="none"/>
         <path d="M50 67 C43.4 65.2 38.6 60.4 37.6 54.6 C44.2 54.8 48.8 59 50 65.2Z" stroke="none"/>
         <path d="M50 67 C56.6 65.2 61.4 60.4 62.4 54.6 C55.8 54.8 51.2 59 50 65.2Z" stroke="none"/>
         <path d="M50 57.4 C44.2 55.8 39.8 51.4 38.8 46 C45 46.2 49.2 50.2 50 55.6Z" stroke="none"/>
         <path d="M50 57.4 C55.8 55.8 60.2 51.4 61.2 46 C55 46.2 50.8 50.2 50 55.6Z" stroke="none"/>
         <path d="M50 47.8 C45 46.2 41.4 42.4 40.6 37.8 C46 38 49.4 41.4 50 46Z" stroke="none"/>
         <path d="M50 47.8 C55 46.2 58.6 42.4 59.4 37.8 C54 38 50.6 41.4 50 46Z" stroke="none"/>
         <circle cx="50" cy="30.6" r="4" stroke="none"/>
         <circle cx="44" cy="35" r="2.9" stroke="none"/>
         <circle cx="56" cy="35" r="2.9" stroke="none"/>
       </g>`;

  $('envSeal').innerHTML = `
    <svg viewBox="0 0 100 100" role="img" aria-label="봉랍">
      <defs>
        <radialGradient id="waxFill" cx="34%" cy="26%" r="80%">
          <stop offset="0" stop-color="#E7CB84"/>
          <stop offset=".52" stop-color="#C8A24A"/>
          <stop offset="1" stop-color="#96722A"/>
        </radialGradient>
        <g id="waxMotif">${motif}</g>
      </defs>
      <circle cx="50" cy="50" r="47" fill="url(#waxFill)"/>
      <circle cx="50" cy="50" r="46.4" fill="none" stroke="#FBEEC6" stroke-opacity=".38" stroke-width="1.2"/>
      <circle cx="50" cy="50" r="39.4" fill="none" stroke="#8A6820" stroke-opacity=".42" stroke-width="2.8"/>
      <circle cx="50" cy="50" r="38.3" fill="none" stroke="#F0DCA6" stroke-opacity=".5" stroke-width="1.3"/>
      <use href="#waxMotif" x="1" y="1.2" style="color:#87661F" opacity=".55"/>
      <use href="#waxMotif" x="-.8" y="-.9" style="color:#F5E3AF" opacity=".7"/>
      <use href="#waxMotif" style="color:#C7A24E"/>
    </svg>`;

  /* ── 개봉 연출 ── */
  document.documentElement.classList.add('intro-lock');

  let finished = false;
  const timers = [];
  const at = (ms, fn) => timers.push(setTimeout(fn, ms));
  const step = cls => intro.classList.add(cls);

  function finish(){
    if (finished) return;
    finished = true;
    timers.forEach(clearTimeout);
    intro.classList.add('is-exit');
    setTimeout(() => {
      intro.classList.add('is-done');
      document.documentElement.classList.remove('intro-lock');
    }, 800);
  }

  at(60,   () => step('is-in'));      // 봉투가 놓인다
  at(1000, () => step('is-break'));   // 봉랍이 떨어진다
  at(1180, () => step('is-open'));    // 덮개가 펼쳐진다
  at(1620, () => step('is-behind'));  // 펼쳐진 덮개가 뒤로
  at(1900, () => step('is-slide'));   // 편지지가 올라온다
  at(2800, finish);                   // 화면으로 전환

  /* 탭하면 바로 넘어갑니다 */
  intro.addEventListener('click', finish);
})();
